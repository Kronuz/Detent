cmake_minimum_required(VERSION 3.24)
project(detent_consumer_check LANGUAGES CXX)
set(DETENT_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Detent source under test")
foreach(option TESTS EXAMPLES BENCHMARKS DISTRIBUTOR MANAGED_AGENT AUTHORITY RELAY PYTHON RUST)
  set(DETENT_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
include(FetchContent)
FetchContent_Declare(detent SOURCE_DIR "${DETENT_SOURCE_ROOT}")
FetchContent_MakeAvailable(detent)
add_executable(detent_consumer_check main.cc)
target_link_libraries(detent_consumer_check PRIVATE detent::detent)
enable_testing()
add_test(NAME detent_consumer_check COMMAND detent_consumer_check)
