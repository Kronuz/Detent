cmake_minimum_required(VERSION 3.16)
project(pin_wave_syscalls LANGUAGES C CXX)
# The oracle scenario (Xapiand's completion-flush-mode review, P5) against one completion source.
# OLD_API: completion before FlushMode; OLD_DEFAULT: with default options, as Detent built it.
set(COMPLETION_SOURCE "" CACHE PATH "completion source")
option(OLD_API "Build against completion before FlushMode" OFF)
option(OLD_DEFAULT "With OLD_API: default PosixIOOptions" OFF)
option(NO_SYNC_DATA "Leave out the SyncData steps (completion before efc368f)" OFF)
set(COMPLETION_TASK OFF CACHE BOOL "" FORCE)
set(COMPLETION_ASIO OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMPLETION_SOURCE}" completion)
add_executable(oracle_scenario oracle_scenario.cc)
target_link_libraries(oracle_scenario PRIVATE completion::completion)
foreach(option OLD_API OLD_DEFAULT NO_SYNC_DATA)
  if(${option})
    target_compile_definitions(oracle_scenario PRIVATE ${option})
  endif()
endforeach()
if(APPLE)
  add_library(interpose SHARED interpose.c)
endif()
